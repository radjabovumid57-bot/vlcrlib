vlcrlib — Cheat Sheet

Быстрый справочник по публичному API. R — right-handed Y-up, метры; Vulkan depth 0..1.

Минимальное приложение

#include <vlcrlib.h>
#include <vloaders.h>   /* FBX/OBJ/USD, LoadModelAuto */
#include <SDL3/SDL.h>

int main(void) {
    if (!InitWindow(1280, 720, "vlcrlib")) return 1;
    SetTargetFPS(60);

    Camera3D cam = { .position={0,1.5f,4}, .target={0,0.5f,0},
                     .up={0,1,0}, .fovy=45,
                     .nearPlane=0.05f, .farPlane=100 };

    while (!WindowShouldClose()) {
        BeginDrawing();
        ClearBackground(RAYWHITE);

        BeginMode3D(cam);
        DrawGrid(10, 1.0f);
        DrawCube((Vector3){0,0.5f,0}, (Vector3){1,1,1}, RED);
        EndMode3D();

        DrawText("Hello, vlcrlib", 20, 20, 20, DARKGRAY);
        EndDrawing();
    }
    CloseWindow();
    return 0;
}
C: compound literals `(Vector3){...}`. C++: `Vector3{...}` / `Camera3D{...}`.

// crossfade idle -> walk

    (Vector3){0}, (Vector3){0}, (Vector3){1,1,1}, WHITE);

// state machine (vanimator.h)
Animator *an = CreateAnimator(sk);
AnimatorAddState(an, "idle", idle, true);
AnimatorAddState(an, "walk", walk, true);
AnimatorPlay(an, "idle", 0.2f);
AnimatorAddTransition(an, "idle", "walk",
                      "speed", ANIM_COND_GREATER, 0.5f, 0.2f);
AnimatorSetFloat(an, "speed", 1.0f);
AnimatorSetOverlay(an, "idle", 0.35f);
// or additive relative to bind pose:
AnimatorSetAdditiveOverlay(an, "breath", 0.5f);
AnimatorAddEvent(an, "walk", 0.5f, "footstep");
// each frame:
const char *ev;
while ((ev = AnimatorNextEvent(an)) != NULL)
    printf("%s\n", ev);
AnimatorSetRootMotion(an, true);
AnimatorUpdate(an, dt);
pos = Vector3Add(pos, AnimatorGetRootMotion(an));
AnimatorDraw(an, pos, (Vector3){0},
             (Vector3){1,1,1}, WHITE);

// pose + IK (bone chain upper->mid->end)
BonePose pose[256];

                     (Vector3){1,1,1}, WHITE);
UnloadSkinnedModel(sk);
int bones = GetSkinnedModelBoneCount(sk);

    printf("%d %s\n", b,

Частые ошибки

  • Чёрный экран 3D: не вызван BeginMode3D; камера внутри объекта; near/far.
  • Модель гигантская: FBX в сантиметрах — масштаб 0.01.
  • Текстура полосами: RGB залит как RGBA / неверный bufferRowLength.
  • GUI без текста: VGuiBeginFrame(...,(Font){0},...).
  • Нет анимации: в FBX нет костей (bones=0) или клип не добавлен.
  • IBL ломает кадр: загрузка внутри BeginDrawing.
  • USD не грузится: не собран OpenUSD, LoadUSD — stub.
