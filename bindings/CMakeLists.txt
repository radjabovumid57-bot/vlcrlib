# ==============================================================================
#  bindings/CMakeLists.txt — native shared libraries behind the language bindings
# ==============================================================================
# Adds:
#   vlcr_ffi   shared library exporting the stable C ABI (bindings/ffi/vlcr_ffi.h)
#   vlcr_jni   JNI bridge used by the Java package (needs a JDK)
#
# Included from the root CMakeLists.txt with -DVLCRLIB_BUILD_BINDINGS=ON.
# ==============================================================================

add_library(vlcr_ffi SHARED ffi/vlcr_ffi.c)
add_library(vlcrlib::ffi ALIAS vlcr_ffi)

# On ELF/Mach-O the static library ends up inside the shared object, so it must
# be position independent.
set_target_properties(vlcrlib PROPERTIES POSITION_INDEPENDENT_CODE ON)

target_link_libraries(vlcr_ffi PRIVATE vlcrlib)
target_include_directories(vlcr_ffi
    PUBLIC  ${CMAKE_CURRENT_SOURCE_DIR}/ffi
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/ffi)

set_target_properties(vlcr_ffi PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    OUTPUT_NAME "vlcr_ffi")

target_compile_definitions(vlcr_ffi PUBLIC VLCR_FFI_USE_SHARED)

if(VLCRLIB_GRAPHICS_CPU)
    # The GPU-free build has no audio backend; the shim compiles the audio
    # entry points to no-ops.
    target_compile_definitions(vlcr_ffi PRIVATE VLCR_FFI_HAS_AUDIO=0)
else()
    target_compile_definitions(vlcr_ffi PRIVATE VLCR_FFI_HAS_AUDIO=1)
endif()

# vlcrlib is static, so its dependencies (SDL3, Vulkan, ...) are linked into
# the shared object. The runtime DLLs are copied next to the bindings by
# scripts/build_bindings.ps1.

# ------------------------------------------------------------------------------
#  C++ (header-only wrapper over the C API; no shared library involved)
# ------------------------------------------------------------------------------
add_library(vlcrlib_cpp INTERFACE)
target_include_directories(vlcrlib_cpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include)
target_link_libraries(vlcrlib_cpp INTERFACE vlcrlib)
target_compile_features(vlcrlib_cpp INTERFACE cxx_std_17)

option(VLCRLIB_BUILD_BINDING_EXAMPLES "Build the language binding examples" ON)
if(VLCRLIB_BUILD_BINDING_EXAMPLES)
    add_executable(vlcr_cpp_hello cpp/examples/hello.cpp)
    target_link_libraries(vlcr_cpp_hello PRIVATE vlcrlib_cpp)
endif()

# ------------------------------------------------------------------------------
#  Java (JNI bridge) — only when a JDK is available
# ------------------------------------------------------------------------------
option(VLCRLIB_BUILD_JNI "Build the JNI bridge used by the Java binding" ON)

set(VLCR_JAVA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/java/src/main/java)
set(VLCR_JNI_SRC ${CMAKE_CURRENT_SOURCE_DIR}/java/native/vlcr_jni.c)

if(VLCRLIB_BUILD_JNI AND EXISTS "${VLCR_JAVA_SRC}" AND EXISTS "${VLCR_JNI_SRC}")
    find_package(Java QUIET COMPONENTS Development)
    if(Java_FOUND)
        if(NOT DEFINED JAVA_HOME)
            get_filename_component(_vlcr_javac "${Java_JAVAC_EXECUTABLE}" REALPATH)
            get_filename_component(_vlcr_javac_dir "${_vlcr_javac}" DIRECTORY)
            get_filename_component(JAVA_HOME "${_vlcr_javac_dir}" DIRECTORY)
            set(JAVA_HOME "${JAVA_HOME}" CACHE PATH "Java installation used for the JNI bridge")
        endif()
        find_package(JNI QUIET)
    endif()

    if(JNI_FOUND)
        file(GLOB_RECURSE VLCR_JAVA_FILES CONFIGURE_DEPENDS "${VLCR_JAVA_SRC}/*.java")
        set(VLCR_JNI_GEN ${CMAKE_CURRENT_BINARY_DIR}/jni-headers)

        # The JNI glue includes the header generated by `javac -h`, so header
        # generation is an explicit step that runs before compilation.
        add_custom_command(
            OUTPUT  ${VLCR_JNI_GEN}/org_vlcrlib_Native.h
            COMMAND ${CMAKE_COMMAND} -E make_directory "${VLCR_JNI_GEN}/classes"
            COMMAND ${Java_JAVAC_EXECUTABLE} -h "${VLCR_JNI_GEN}"
                    -d "${VLCR_JNI_GEN}/classes" ${VLCR_JAVA_FILES}
            DEPENDS ${VLCR_JAVA_FILES}
            COMMENT "Generating JNI headers for the Java binding"
            VERBATIM)

        add_custom_target(vlcr_jni_headers DEPENDS ${VLCR_JNI_GEN}/org_vlcrlib_Native.h)

        add_library(vlcr_jni SHARED ${VLCR_JNI_SRC})
        add_dependencies(vlcr_jni vlcr_jni_headers vlcr_ffi)
        target_include_directories(vlcr_jni PRIVATE ${JNI_INCLUDE_DIRS} "${VLCR_JNI_GEN}")
        target_link_libraries(vlcr_jni PRIVATE vlcr_ffi)
        set_target_properties(vlcr_jni PROPERTIES
            POSITION_INDEPENDENT_CODE ON
            OUTPUT_NAME "vlcr_jni")
        message(STATUS "vlcrlib: JNI bridge enabled (JAVA_HOME=${JAVA_HOME})")
    else()
        message(STATUS "vlcrlib: JNI bridge skipped (no JDK found; set JAVA_HOME and re-run CMake)")
    endif()
endif()
